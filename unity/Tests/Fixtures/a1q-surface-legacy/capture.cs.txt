using System;
using System.IO;
using System.Linq;
using System.Reflection;
using NUnit.Framework;
using Spacewars.Runtime;
using Spacewars.Simulation;
namespace Spacewars.Tests.EditMode
{
    public sealed class LegacySurfaceCaptureTests
    {
        [Test] public void CaptureExactBaselineWorlds()
        {
            string output=Environment.GetEnvironmentVariable("SURFACE_LEGACY_OUTPUT");Assert.IsNotEmpty(output);Directory.CreateDirectory(output);
            var type=typeof(PlayableRuntime).Assembly.GetType("Spacewars.Runtime.PlayableDomain",true);var flags=BindingFlags.Instance|BindingFlags.NonPublic;
            foreach(var map in new[]{"flat","crossings","foundry"}){
                var profile=map=="flat"?PlayableProfile.Default:map=="crossings"?PlayableProfile.ThreeCrossingsDefault:PlayableProfile.Create(PlayableProfile.Default.CopyData(),new FoundryMap(new FoundryProfileData()));
                var config=map=="foundry"?((FoundryMap)profile.AuthoredMap).Configuration(profile,19092026):null;
                var domain=config==null?Activator.CreateInstance(type,flags,null,new object[]{profile,1L,false},null):Activator.CreateInstance(type,flags,null,new object[]{profile,1L,config},null);
                object Call(string name,params object[] args)=>type.GetMethod(name,flags).Invoke(domain,args);
                var nav=(NavigationSession)type.GetProperty("Navigation",flags).GetValue(domain);
                var actor=nav.Crowd.Units.First(u=>u.Team==(config==null?0:config.Roster[0].Team));var goal=new NavPoint(actor.Position.X,actor.Position.Z+(map=="flat"?2:-2));
                Assert.AreEqual(PlayableCommandStatus.Applied,Call("Apply",new PlayableCommand(1,1,config?.Roster[0].Id??"player-1",PlayableCommandKind.Move,new[]{actor.Id},goal).AsHuman(),null));
                void Save(string stage)=>File.WriteAllBytes(Path.Combine(output,map+"-"+stage+".world"),(byte[])Call("CaptureWorldBytes",19092026,config?.SourceIdentity??"a1q-surface-legacy"));
                Save("pending");
                while(nav.Requests.TryDequeue(out var r))nav.Answers.TryEnqueue(new NavigationAnswer(r,new SharedFlowRouter(r.Geometry,r.Profile).FindPath(r.Start,r.Goal)));
                nav.ApplyResults();Assert.True(actor.Moving);Save("installed");
                for(int tick=0;tick<300;tick++)Call("Step",1d/30);
                Assert.AreEqual(NavigationOutcome.Arrived,actor.Outcome);Save("arrived");
                if(map=="flat")Assert.NotNull(Call("SpatialCompletion",actor.Id,PlayableOwner.Player));
            }
        }
    }
}
